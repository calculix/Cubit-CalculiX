add_library(cubitsim_system STATIC ProcessIdentity.hpp SystemResourceReader.hpp)
if (WIN32)
  target_sources(cubitsim_system PRIVATE SystemResourceReaderWindows.cpp)
  target_compile_definitions(cubitsim_system PRIVATE _WIN32_WINNT=0x0601 PSAPI_VERSION=1)
  target_link_libraries(cubitsim_system PRIVATE psapi pdh)
elseif (CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_sources(cubitsim_system PRIVATE SystemResourceReaderLinux.cpp)
else ()
  message(FATAL_ERROR "cubit-sim resource monitoring supports Windows and Linux.")
endif ()
target_compile_features(cubitsim_system PUBLIC cxx_std_17)
target_include_directories(cubitsim_system PUBLIC "${CUBITSIM_SOURCE_ROOT}")
set_target_properties(cubitsim_system PROPERTIES POSITION_INDEPENDENT_CODE ON)
