# Shared units and materials. This target must never depend on workspace modules.
add_library(cubitsim_model SHARED ../ModelExport.hpp ../Interfaces/ModelEditorInterface.cpp ../Interfaces/ModelEditorInterface.hpp)
target_compile_features(cubitsim_model PUBLIC cxx_std_17)
target_compile_definitions(cubitsim_model PRIVATE CUBITSIM_MODEL_BUILD)
target_include_directories(cubitsim_model PUBLIC "${CUBITSIM_SOURCE_ROOT}")
target_link_libraries(cubitsim_model PUBLIC cubitsim_storage PRIVATE cubit_util Qt6::Core)
add_subdirectory(units)
add_subdirectory(materials)
option(CUBITSIM_BUILD_MODEL_TESTS "Build Framework unit and material service regression checks" OFF)
if (CUBITSIM_BUILD_MODEL_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
set_target_properties(cubitsim_model PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CUBITSIM_DEPLOY_DIRECTORY}" LIBRARY_OUTPUT_DIRECTORY "${CUBITSIM_DEPLOY_DIRECTORY}" ARCHIVE_OUTPUT_DIRECTORY "${CUBITSIM_DEPLOY_DIRECTORY}" BUILD_RPATH_USE_ORIGIN TRUE)
if (UNIX)
  set_target_properties(cubitsim_model PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN;$ORIGIN/../../lib" BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH_USE_LINK_PATH FALSE)
endif()
