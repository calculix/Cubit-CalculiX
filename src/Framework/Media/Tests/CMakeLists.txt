cmake_minimum_required(VERSION 3.20)
project(CubitSimMediaTests LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Core Gui)
find_package(Threads REQUIRED)
get_filename_component(CUBITSIM_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(CUBITSIM_DEPLOY_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>")
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/.." media)
enable_testing()
add_executable(cubitsim_media_tests MediaTests.cpp)
target_link_libraries(cubitsim_media_tests PRIVATE cubitsim_media)
target_compile_features(cubitsim_media_tests PRIVATE cxx_std_17)
set_target_properties(cubitsim_media_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CUBITSIM_DEPLOY_DIRECTORY}")
add_test(NAME cubitsim_media COMMAND cubitsim_media_tests)
